/**
 * TELL US ABOUT THE WATER – backend van de enquête (Club Copains)
 * Ontvangt de antwoorden van de enquêtepagina, zet ze in een Google-sheet en maakt de analyses.
 *
 * EENMALIG INSTELLEN
 * 1. Ga naar script.google.com (best met events@clubcopains.be: bevestigingsmails vertrekken van dat adres).
 *    "Nieuw project", verwijder de voorbeeldcode, plak deze volledige code en sla op.
 * 2. Kies bovenaan de functie  installeer  en klik "Uitvoeren". Geef toestemming
 *    (de melding "Google heeft deze app niet geverifieerd" is normaal: Geavanceerd > Ga naar ...).
 *    In het logboek staat daarna de link naar de resultatensheet.
 * 3. Klik rechtsboven "Implementeren" > "Nieuwe implementatie" > type "Web-app".
 *    Uitvoeren als: Ik. Wie heeft toegang: Iedereen. Klik "Implementeren" en kopieer de web-app-URL.
 * 4. Die URL komt in de enquêtepagina (SCRIPT_URL). Daarna komen antwoorden automatisch binnen.
 *
 * TESTEN
 * - testAntwoorden          : zet 60 willekeurige testantwoorden in de sheet en maakt de analyses.
 * - verwijderTestAntwoorden : wist ALLE antwoorden en e-mailadressen (enkel vóór de echte lancering).
 * - vernieuwAnalyse         : herberekent de analyses (gebeurt ook automatisch elke 10 minuten).
 *
 * DASHBOARD
 * - Na het implementeren: voer  toonDashboardLink  uit. In het logboek staat je privélink naar het dashboard
 *   (vraag 1: gaan mensen minder uit? vraag 2: wat houdt hen tegen? vraag 3: tickets; extra). Deel die link niet publiek.
 *
 * TABBLADEN IN DE SHEET
 * - Overzicht en Per leeftijd/gender/woonplaats/situatie : alle vragen in procenten, om op te zoeken.
 * - Antwoorden   : de ruwe antwoorden, één rij per persoon (Bestand > Downloaden > Excel).
 * - E-mails      : e-mailadressen voor de resultaten, apart van de antwoorden.
 *
 * KWALITEITSCONTROLE (kolom "Kwaliteit")
 * - OK                        : telt mee in de analyses.
 * - Te snel ingevuld          : minder dan MIN_SECONDEN over de hele enquête. Telt niet mee.
 * - Spam                      : een verborgen veld dat mensen niet zien, werd ingevuld (typisch voor bots). Telt niet mee.
 * - Mogelijk 2e keer          : dezelfde browser vulde de enquête al eens in. Telt niet mee.
 * Wie een e-mailadres geeft, krijgt een bevestigingsmail. Alleen bevestigde adressen krijgen de resultaten.
 */

// ========== HULPJES OM VRAGEN TE DEFINIËREN ==========
function mc(title, options, help, required) { return { type: 'mc', title, options, help: help || '', required: required !== false }; }
function cb(title, options, max, help) { return { type: 'cb', title, options, max: max || 0, help: help || (max ? MAX_HELP(max) : 'Multiple answers possible.'), required: true }; }
// Zelfde, maar met een vrij invulveld "Other" voor eigen redenen en ideeën
function cbx(title, options, max, help) { const q = cb(title, options, max, help); q.other = true; return q; }
function schaal(title, laag, hoog, help) { return { type: 'scale', title, laag, hoog, help: help || '', required: true }; }
function openVraag(title, help) { return { type: 'para', title, help: help || '', required: false }; }
function mailVraag(title, help) { return { type: 'email', title, help: help || '', required: false }; }

const KOL_TIJD = 'Tijdstip', KOL_BEVESTIGD = 'E-mail bevestigd', KOL_DUUR = 'Duur (seconden)',

// Alle vragen waarvan we procenten berekenen (geen open vragen en geen e-mail)
function vraagLijst() {
  const L = [];
  const zet = (q) => {
    if (q.type === 'mc') L.push({ titel: q.title, opties: q.options.slice(), meer: false });
    else if (q.type === 'cb') L.push({ titel: q.title, opties: q.options.slice(), meer: true });
    else if (q.type === 'scale') L.push({ titel: q.title, opties: ['1', '2', '3', '4', '5'], schaal: true });
  };
  ALGEMEEN.forEach(zet);
  zet(DREMPELS);
  const labels = REDENEN.map(r => r.label);
  L.push({ titel: TITEL_ZWAARST, opties: labels });
  L.push({ titel: TITEL_NR2, opties: labels });
  L.push({ titel: TITEL_NR3, opties: labels });
  L.push({ titel: TITEL_EXTRA, opties: labels, meer: true });
  REDENEN.forEach(r => DIEPTE[r.key].vragen.forEach(zet));
  TICKETS.forEach(zet);
  NA.forEach(zet);
  zet(EINDE[0]);
  return L;
}

function kolommen() {
  const open = q => q.type === 'para';
  const titels = [].concat(
    ALGEMEEN.map(q => q.title), [DREMPELS.title, TITEL_ZWAARST, TITEL_NR2, TITEL_NR3, TITEL_EXTRA],
    ...REDENEN.map(r => DIEPTE[r.key].vragen.map(q => q.title)),
    TICKETS.map(q => q.title), NA.map(q => q.title), [EINDE[0].title]);
  return [KOL_TIJD].concat(titels, [KOL_BEVESTIGD, KOL_DUUR, KOL_KWALITEIT, KOL_SESSIE]);
}

function installeer() {
  const ss = SpreadsheetApp.create(TITEL + ' – resultaten');
  ss.getSheets()[0].setName('Overzicht');
  ['Per leeftijd', 'Per gender', 'Per woonplaats', 'Per situatie'].forEach(n => ss.insertSheet(n));
  const a = ss.insertSheet('Antwoorden');
  const k = kolommen();
  a.getRange(1, 1, 1, k.length).setValues([k]).setFontWeight('bold').setBackground('#e8eefc').setWrap(true);
  a.setFrozenRows(1);
  a.getRange(2, 2, a.getMaxRows() - 1, k.length - 1).setNumberFormat('@');   // antwoorden als tekst bewaren
  const m = ss.insertSheet('E-mails');

function veilig(v) {
  let s = Array.isArray(v) ? v.map(String).join(' | ') : (v == null ? '' : String(v));
  s = s.slice(0, 1000);
  return /^[=+\-@]/.test(s) ? "'" + s : s;   // geen formules laten uitvoeren
}

function verwerk(d, mailen) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const ss = sheet();
    const duur = Number(d._duur) || 0;
    const kwaliteit = d._val ? 'Spam' : duur < MIN_SECONDEN ? 'Te snel ingevuld' : d._herhaald === 'ja' ? 'Mogelijk 2e keer' : 'OK';
    const email = String(d[TITEL_EMAIL] || '').trim();
    const geldig = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(email);
    const sessie = String(d._id || Utilities.getUuid()).slice(0, 40);
    const rij = kolommen().map(k => {
      if (k === KOL_TIJD) return new Date();
      if (k === KOL_BEVESTIGD) return email && geldig && kwaliteit !== 'Spam' ? 'Nog niet' : '';
      if (k === KOL_DUUR) return duur;
      if (k === KOL_KWALITEIT) return kwaliteit;
      if (k === KOL_SESSIE) return sessie;
      return veilig(d[k]);
    });
    ss.getSheetByName('Antwoorden').appendRow(rij);

    if (email && geldig && kwaliteit !== 'Spam') {
      const token = Utilities.getUuid();
      ss.getSheetByName('E-mails').appendRow([new Date(), email, 'Nee', token, sessie]);
      if (mailen) stuurBevestiging(email, token);
    }
  } finally {
    lock.releaseLock();
  }
}

    '<p>Didn\'t fill in our survey? Then you can simply ignore this email.</p>' +

function beantwoord(a) { return a != null && a !== '' && !(Array.isArray(a) && a.length === 0); }
function heeft(a, optie) { return beantwoord(a) && (Array.isArray(a) ? a.indexOf(optie) > -1 : String(a) === optie); }
function blad(ss, naam) { return ss.getSheetByName(naam) || ss.insertSheet(naam); }
function nu() { return Utilities.formatDate(new Date(), 'Europe/Brussels', 'dd/MM/yyyy HH:mm'); }

// Bouwt een tabel op en schrijft ze weg met opmaak
function Tabel(breedte) {
  this.W = breedte; this.rijen = []; this.fmt = []; this.koppen = [];
}
Tabel.prototype.rij = function (r, f) {
  r = r.slice(); while (r.length < this.W) r.push('');
  this.rijen.push(r);
  this.fmt.push(r.map((_, i) => i === 0 ? '@' : (f || '0%')));
};
Tabel.prototype.kop = function (r) { this.koppen.push(this.rijen.length + 1); this.rij(r, '@'); };
Tabel.prototype.schrijf = function (sh) {
  sh.clear();
  sh.getCharts().forEach(c => sh.removeChart(c));
  const rng = sh.getRange(1, 1, this.rijen.length, this.W);
  rng.setNumberFormats(this.fmt);
  rng.setValues(this.rijen);
  this.koppen.forEach(r => sh.getRange(r, 1, 1, this.W).setFontWeight('bold').setBackground('#e8eefc'));
  sh.getRange(1, 1).setFontWeight('bold').setFontSize(12);
  sh.setColumnWidth(1, 420);
  sh.getRange(1, 1, this.rijen.length, 1).setWrap(true);
  sh.setFrozenColumns(1);
};

function groeperen(antw, dim) {
  const groepen = ['Totaal'].concat(dim.opties);
  return { groepen, per: groepen.map(g => g === 'Totaal' ? antw : antw.filter(a => a[dim.titel] === g)) };
}

function maakKruistabel(ss, dim, vragen, antw, per) {
  const d = per[dim.vraag];
  const { groepen, per: groep } = groeperen(antw, d);
  const t = new Tabel(groepen.length + 1);
  t.rij(['Analyse per ' + dim.naam + ' (% van wie de vraag beantwoordde) – bijgewerkt ' + nu()], '@');
  t.kop(['Aantal deelnemers'].concat(groepen));
  t.rij([''].concat(groep.map(l => l.length)), '0');
  t.rij([]);
  vragen.forEach(v => {
    if (v.titel === d.titel) return;
    const id = v.titel;
    const basis = groep.map(l => l.filter(a => beantwoord(a[id])));
    t.kop([v.titel].concat(groepen));
    v.opties.forEach(o => t.rij([o].concat(basis.map(l => l.length ? l.filter(a => heeft(a[id], o)).length / l.length : ''))));
    if (v.schaal) t.rij(['Gemiddelde (1-5)'].concat(basis.map(l => l.length ? l.reduce((s, a) => s + Number(a[id]), 0) / l.length : '')), '0.0');
    t.rij(['Aantal antwoorden (n)'].concat(basis.map(l => l.length)), '0');
    t.rij([]);
  });
  t.schrijf(blad(ss, dim.blad));
}

function rangschik(v, groep) {
  const id = v.titel;
  const basis = groep.map(l => l.filter(a => beantwoord(a[id])));
  return v.opties
    .map(o => [o].concat(basis.map(l => l.length ? l.filter(a => heeft(a[id], o)).length / l.length : '')))
    .sort((x, y) => (Number(y[1]) || 0) - (Number(x[1]) || 0));
}

function maakOverzicht(ss, antw, per, alle) {
  const { groepen, per: groep } = groeperen(antw, per[TITEL_LEEFTIJD]);
  const t = new Tabel(groepen.length + 1);
  t.rij(['OVERZICHT per leeftijd – bijgewerkt ' + nu() + ' (enkel antwoorden met kwaliteit OK)'], '@');
  t.kop(['Aantal deelnemers'].concat(groepen));
  t.rij([''].concat(groep.map(l => l.length)), '0');
  t.rij([]);
  t.kop(['Kwaliteitscontrole (alle binnengekomen antwoorden)', 'Aantal']);
  ['OK', 'Te snel ingevuld', 'Mogelijk 2e keer', 'Spam'].forEach(k => t.rij([k, alle.filter(x => x[KOL_KWALITEIT] === k).length], '0'));
  t.rij(['Bevestigde e-mailadressen', alle.filter(x => x[KOL_BEVESTIGD] === 'Ja').length], '0');
  t.rij([]);

  // Verschuiving uitgaansfrequentie
  const v = groep.map(l => {
    const b = l.filter(a => FREQ.indexOf(a[TITEL_VROEGER]) > -1 && FREQ.indexOf(a[TITEL_NU]) > -1);
    const minder = b.filter(a => FREQ.indexOf(a[TITEL_NU]) > FREQ.indexOf(a[TITEL_VROEGER])).length;
    const gelijk = b.filter(a => FREQ.indexOf(a[TITEL_NU]) === FREQ.indexOf(a[TITEL_VROEGER])).length;
    return { n: b.length, minder, gelijk, meer: b.length - minder - gelijk };
  });
  t.kop(['Uitgaan nu t.o.v. 2 à 3 jaar geleden'].concat(groepen));
  t.rij(['Gaat minder uit'].concat(v.map(x => x.n ? x.minder / x.n : '')));
  t.rij(['Gaat even vaak uit'].concat(v.map(x => x.n ? x.gelijk / x.n : '')));
  t.rij(['Gaat meer uit'].concat(v.map(x => x.n ? x.meer / x.n : '')));
  t.rij([]);

  // Top 3, nummer 1 en wat er verder nog meespeelt
  const startDrempels = t.rijen.length + 1;
  const drempels = rangschik(per[TITEL_DREMPELS], groep);
  t.kop(['Top 3: wat houdt mensen tegen? (% dat dit aanvinkte)'].concat(groepen));
  drempels.forEach(r => t.rij(r));
  t.rij([]);
  t.kop(['Nummer 1 (% van alle deelnemers)'].concat(groepen));
  rangschik(per[TITEL_ZWAARST], groep).forEach(r => t.rij(r));
  t.rij([]);
  t.kop(['Speelt daarnaast ook mee (% van wie de vraag invulde)'].concat(groepen));
  rangschik(per[TITEL_EXTRA], groep).forEach(r => t.rij(r));

  const sh = blad(ss, 'Overzicht');
  t.schrijf(sh);
  try {
    const chart = sh.newChart().setChartType(Charts.ChartType.BAR)
      .addRange(sh.getRange(startDrempels, 1, drempels.length + 1, 2))
      .setNumHeaders(1)
      .setOption('title', 'Top 3: wat houdt mensen tegen? (alle deelnemers)')
      .setOption('legend', { position: 'none' })
      .setOption('width', 700).setOption('height', 420)
      .setPosition(startDrempels, groepen.length + 3, 0, 0)
      .build();
    sh.insertChart(chart);
  } catch (e) {}
}

function vraagDef(titel) { return [].concat(...REDENEN.map(r => DIEPTE[r.key].vragen)).filter(q => q.title === titel)[0]; }
// Antwoorden die echt doorwegen: minstens 20% (min. 1, max. 5)
function sterk(d) { if (!d || !d.alles) return []; const s = d.alles.filter(x => x.p >= 0.2).slice(0, 5); return s.length ? s : d.alles.slice(0, 1); }

function pctVan(lijst, titel, optie) {
  const b = lijst.filter(a => beantwoord(a[titel]));
  return b.length ? b.filter(a => heeft(a[titel], optie)).length / b.length : 0;
}
function rangLijst(lijst, q, max) {
  const b = lijst.filter(a => beantwoord(a[q.title]));
  const opties = q.type === 'scale' ? ['1', '2', '3', '4', '5'] : q.options;
  const top = opties.map(o => ({ o, p: b.length ? b.filter(a => heeft(a[q.title], o)).length / b.length : 0 }))
    .sort((x, y) => y.p - x.p);
  const uit = { vraag: q.title, n: b.length, top: max ? top.slice(0, max) : top, alles: top };
  if (q.type === 'scale') { uit.schaal = true; uit.gem = b.length ? b.reduce((s, a) => s + Number(a[q.title]), 0) / b.length : 0; uit.laag = q.laag; uit.hoog = q.hoog; }
  return uit;
}
function per0(titel) { return ALGEMEEN.filter(q => q.title === titel)[0].options; }

function berekenDashboard(antw, alle) {
  const S = { n: antw.length, binnen: alle.length, bijgewerkt: nu(), minN: MIN_N };

  // 1. Gaan mensen minder uit?
  const verg = antw.filter(a => FREQ.indexOf(a[TITEL_VROEGER]) > -1 && FREQ.indexOf(a[TITEL_NU]) > -1);
  const vs = a => FREQ.indexOf(a[TITEL_NU]) - FREQ.indexOf(a[TITEL_VROEGER]);
  const telling = l => ({ n: l.length, aMinder: l.filter(a => vs(a) > 0).length, aEven: l.filter(a => vs(a) === 0).length, aMeer: l.filter(a => vs(a) < 0).length, minder: l.length ? l.filter(a => vs(a) > 0).length / l.length : 0,
    even: l.length ? l.filter(a => vs(a) === 0).length / l.length : 0, meer: l.length ? l.filter(a => vs(a) < 0).length / l.length : 0 });
  S.uitgaan = { rijen: LEEFTIJD.map(g => Object.assign({ groep: g }, telling(verg.filter(a => a[TITEL_LEEFTIJD] === g)))).filter(r => r.n),
                totaal: telling(verg) };
  const kandidaten = S.uitgaan.rijen.filter(r => r.n >= MIN_N_GROEP).sort((x, y) => y.minder - x.minder);
  S.uitgaan.grootst = kandidaten[0] || null;
  S.uitgaan.kleinst = kandidaten.length > 1 ? kandidaten[kandidaten.length - 1] : null;
  const afhakers = verg.filter(a => vs(a) > 0), blijvers = verg.filter(a => vs(a) <= 0);
  S.afhakers = afhakers.length; S.blijvers = blijvers.length;

  // 2. Wat houdt mensen tegen?
  S.redenen = REDENEN.map(r => {
    const nr1 = antw.length ? antw.filter(a => a[TITEL_ZWAARST] === r.label).length / antw.length : 0;
    const top3 = pctVan(antw, DREMPELS.title, r.label);
    const nr1Afh = afhakers.length ? afhakers.filter(a => a[TITEL_ZWAARST] === r.label).length / afhakers.length : 0;
    const nr1Bl = blijvers.length ? blijvers.filter(a => a[TITEL_ZWAARST] === r.label).length / blijvers.length : 0;
    const details = DIEPTE[r.key].vragen.filter(q => q.type !== 'para').map(q => rangLijst(antw, q, 3)).filter(d => d.n);

      const g = per0(dim).map(o => { const l = antw.filter(a => a[dim] === o); return { o, n: l.length, p: pctVan(l, DREMPELS.title, r.label) }; })
        .filter(x => x.n >= MIN_N_GROEP).sort((x, y) => y.p - x.p);
      if (g.length > 1 && g[0].p - g[g.length - 1].p >= VERSCHIL) verschillen.push({ kenmerk: naam, hoog: g[0], laag: g[g.length - 1] });
    });
    const tel = {};
    DIEPTE[r.key].vragen.filter(q => q.other).forEach(q => antw.forEach(a => (Array.isArray(a[q.title]) ? a[q.title] : []).forEach(v => {
      if (q.options.indexOf(v) > -1) return;
      const k = String(v).toLowerCase().replace(/[^a-z0-9à-ÿ ]/g, '').replace(/\s+/g, ' ').trim();

    })));
    const voorstellen = Object.keys(tel).filter(k => tel[k] >= 3).sort((x, y) => tel[y] - tel[x]).slice(0, 5).map(k => ({ v: k, aantal: tel[k] }));
    const rel = nr1 >= 0.1 || top3 >= 0.2 ? 'Doorslaggevend' : nr1 >= 0.08 || top3 >= 0.17 ? 'Op de rand' : top3 >= 0.05 ? 'Speelt mee' : 'Verwaarloosbaar';
    const kern = sterk(rangLijst(antw, vraagDef(KERN[r.key]))).slice(0, 2);

    return { label: r.label, titel: DIEPTE[r.key].titel, nr1, top3, nr1Afh, nr1Bl, rel, details, verschillen, voorstellen, kern, oplos };
  }).sort((x, y) => y.nr1 - x.nr1 || y.top3 - x.top3);
  S.niets = pctVan(antw, DREMPELS.title, NIETS);

  // 3. Tickets
  const tw = TICKETS[0].title;

    aWeek: wanneer.filter(a => a[tw] === 'In the week of the party').length, aLaatst: wanneer.filter(a => LAATST.indexOf(a[tw]) > -1).length, aVroeg: wanneer.filter(a => VROEG.indexOf(a[tw]) > -1).length,

  };
  S.tickets.waaromSterk = sterk(S.tickets.waarom); S.tickets.helptSterk = sterk(S.tickets.helpt); S.tickets.kortingTop = S.tickets.korting.alles[0];
  // Groepen per gender en situatie voor vraag 1 (extra)
  S.uitgaan.gender = per0(TITEL_GENDER).map(g => Object.assign({ groep: g }, telling(verg.filter(a => a[TITEL_GENDER] === g)))).filter(r => r.n);
  S.uitgaan.situatie = per0(TITEL_DAG).map(g => Object.assign({ groep: g }, telling(verg.filter(a => a[TITEL_DAG] === g)))).filter(r => r.n);

  // 4. Extra
  S.extra = NA.filter(q => q.type !== 'para').map(q => rangLijst(antw, q, 3)).filter(d => d.n);
  return S;
}

  const per = {}; vraagLijst().forEach(v => { per[v.titel] = v; });

}

function testAntwoorden() {
  const kies = l => l[Math.floor(Math.random() * l.length)];
  const schud = l => l.slice().sort(() => Math.random() - 0.5);
  const geef = (d, q) => {
    if (q.type === 'mc') d[q.title] = kies(q.options);
    else if (q.type === 'cb') d[q.title] = schud(q.options).slice(0, 1 + Math.floor(Math.random() * Math.min(q.max || 3, q.options.length)));
    else if (q.type === 'scale') d[q.title] = String(1 + Math.floor(Math.random() * 5));
  };
  for (let i = 0; i < 60; i++) {
    const d = { _duur: 120 + Math.floor(Math.random() * 400), _id: 'test' + i, _val: '', _herhaald: 'nee' };
    ALGEMEEN.forEach(q => geef(d, q));
    const gekozen = schud(REDENEN).slice(0, 1 + Math.floor(Math.random() * 3));
    d[DREMPELS.title] = gekozen.map(r => r.label);
    [TITEL_ZWAARST, TITEL_NR2, TITEL_NR3].forEach((t, k) => { if (gekozen[k]) d[t] = gekozen[k].label; });
    gekozen.forEach(r => DIEPTE[r.key].vragen.forEach(q => geef(d, q)));
    TICKETS.forEach(q => geef(d, q));
    NA.forEach(q => geef(d, q));
    verwerk(d, false);
  }
  vernieuwAnalyse();
}
